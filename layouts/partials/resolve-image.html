{{- $page := .page -}} {{- $src := .src -}} {{- if not $src -}} {{- errorf "resolve-image: empty image path" -}} {{- end -}} {{- $src = strings.TrimPrefix "/" $src -}} {{- $img := false -}} {{- if hasPrefix $src "uploads/" -}} {{- $img = resources.Get $src -}} {{- else -}} {{- $img = $page.Resources.GetMatch $src -}} {{- end -}} {{- if not $img -}} {{- errorf "resolve-image: %q not found for %s" $src $page.RelPermalink -}} {{- end -}} {{- return $img -}}
